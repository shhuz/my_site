## 项目 1：温度转换器

摄氏度和华氏度互相转换。

```python
def c_to_f(c):
    """摄氏度 → 华氏度"""
    return c * 9 / 5 + 32

def f_to_c(f):
    """华氏度 → 摄氏度"""
    return (f - 32) * 5 / 9

choice = input("转换方向（1=摄氏转华氏, 2=华氏转摄氏）：")

if choice == "1":
    c = float(input("请输入摄氏度："))
    print(f"{c}°C = {c_to_f(c)}°F")
elif choice == "2":
    f = float(input("请输入华氏度："))
    print(f"{f}°F = {f_to_c(f)}°C")
else:
    print("无效选择")
```

**用到**：函数、`if/else`、`float()` 转换、f-string。

## 项目 2：学生成绩管理（列表 + 循环）

记录成绩、计算平均分、找最高最低。

```python
scores = []   # 空列表存成绩

while True:
    s = input("输入成绩（输入 q 结束）：")
    if s == "q":
        break
    scores.append(float(s))

if scores:   # 列表非空才计算
    print(f"共 {len(scores)} 个成绩")
    print(f"平均分：{sum(scores) / len(scores):.2f}")
    print(f"最高分：{max(scores)}")
    print(f"最低分：{min(scores)}")
else:
    print("没有输入任何成绩")
```

**用到**：列表、`while`、`break`、内置函数 `sum/max/min/len`。

## 项目 3：简易记账本（字典 + 循环）

记录收入和支出，统计结余。

```python
records = []   # 存每笔记录 (类型, 金额, 说明)

while True:
    print("\n1. 记收入  2. 记支出  3. 查看  4. 退出")
    choice = input("选择：")

    if choice == "1":
        amount = float(input("收入金额："))
        desc = input("收入来源：")
        records.append(("收入", amount, desc))
    elif choice == "2":
        amount = float(input("支出金额："))
        desc = input("支出用途：")
        records.append(("支出", amount, desc))
    elif choice == "3":
        total = 0
        for rtype, amount, desc in records:
            print(f"{rtype}：{amount:.2f} 元（{desc}）")
            total += amount if rtype == "收入" else -amount
        print(f"当前结余：{total:.2f} 元")
    elif choice == "4":
        print("再见！")
        break
    else:
        print("无效选择")
```

**用到**：列表、元组解包、循环、条件、`if __name__` 前的函数式思路。

## 项目 4：文件批量重命名（pathlib + 实战）

把某目录下的文件统一改名（现代写法，用 `pathlib`）。

```python
from pathlib import Path

# 改成你要操作的目录
dir_path = Path("./我的文件")

if not dir_path.exists():
    print("目录不存在")
    exit()

# 列出所有 .txt 文件
files = list(dir_path.glob("*.txt"))
print(f"找到 {len(files)} 个 .txt 文件")

for i, f in enumerate(files, start=1):
    new_name = dir_path / f"文件_{i}.txt"   # 重命名为 文件_1.txt ...
    f.rename(new_name)
    print(f"重命名: {f.name} → {new_name.name}")
```

**用到**：`pathlib`、`glob`、`enumerate`、目录操作（现代写法）。

## 项目 5：用类写一个"银行账户"

面向对象实战。

```python
class BankAccount:
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.balance = balance

    def deposit(self, amount):
        self.balance += amount
        print(f"存入 {amount} 元，余额 {self.balance} 元")

    def withdraw(self, amount):
        if amount > self.balance:
            print("余额不足")
        else:
            self.balance -= amount
            print(f"取出 {amount} 元，余额 {self.balance} 元")

    def show(self):
        print(f"{self.owner} 的余额：{self.balance} 元")


acc = BankAccount("小明", 100)
acc.deposit(50)    # 存入 50 元，余额 150 元
acc.withdraw(30)   # 取出 30 元，余额 120 元
acc.show()         # 小明 的余额：120 元
```

**用到**：`class`、`__init__`、属性、方法、面向对象。

## 项目 6：命令行任务管理器（综合实战）

把基础 + 进阶知识全串起来，做一个真实可用的工具。覆盖：`class`、装饰器、生成器、lambda、`argparse`、文件读写、异常。

```python
# todo.py
import json
from pathlib import Path
import argparse
from datetime import datetime

DATA_FILE = Path(__file__).parent / "tasks.json"


class Task:
    """单个任务"""

    def __init__(self, id, title, done=False):
        self.id = id
        self.title = title
        self.done = done

    def to_dict(self):
        return {"id": self.id, "title": self.title, "done": self.done}

    @classmethod
    def from_dict(cls, data):
        return cls(data["id"], data["title"], data["done"])

    def __str__(self):
        status = "[x]" if self.done else "[ ]"
        return f"{status} {self.id}. {self.title}"


def log(func):
    """装饰器：记录操作日志"""
    def wrapper(*args, **kwargs):
        result = func(*args, **kwargs)
        print(f"[{datetime.now():%H:%M}] {func.__name__} 完成")
        return result
    return wrapper


class TaskManager:
    """任务管理器：增删改查 + 持久化"""

    def __init__(self):
        self.tasks = []
        self._load()

    def _load(self):
        if DATA_FILE.exists():
            with open(DATA_FILE, encoding="utf-8") as f:
                data = json.load(f)
                self.tasks = [Task.from_dict(t) for t in data]

    def _save(self):
        with open(DATA_FILE, "w", encoding="utf-8") as f:
            json.dump([t.to_dict() for t in self.tasks], f,
                      ensure_ascii=False, indent=2)

    @log
    def add(self, title):
        new_id = (self.tasks[-1].id + 1) if self.tasks else 1
        self.tasks.append(Task(new_id, title))
        self._save()

    @log
    def remove(self, task_id):
        self.tasks = [t for t in self.tasks if t.id != task_id]
        self._save()

    @log
    def done(self, task_id):
        for t in self.tasks:
            if t.id == task_id:
                t.done = True
                break
        self._save()

    def show(self):
        if not self.tasks:
            print("暂无任务")
            return
        for t in sorted(self.tasks, key=lambda t: t.id):
            print(t)
        total = len(self.tasks)
        finished = sum(1 for t in self.tasks if t.done)
        print(f"\n共 {total} 项，已完成 {finished} 项")


def main():
    parser = argparse.ArgumentParser(description="命令行任务管理器")
    sub = parser.add_subparsers(dest="command", required=True)

    sub.add_parser("list", help="查看所有任务")
    p_add = sub.add_parser("add", help="新增任务")
    p_add.add_argument("title", help="任务内容")
    p_done = sub.add_parser("done", help="标记完成")
    p_done.add_argument("id", type=int, help="任务 ID")
    p_rm = sub.add_parser("remove", help="删除任务")
    p_rm.add_argument("id", type=int, help="任务 ID")

    args = parser.parse_args()
    mgr = TaskManager()

    try:
        if args.command == "list":
            mgr.show()
        elif args.command == "add":
            mgr.add(args.title)
            mgr.show()
        elif args.command == "done":
            mgr.done(args.id)
            mgr.show()
        elif args.command == "remove":
            mgr.remove(args.id)
            mgr.show()
    except Exception as e:
        print(f"操作失败：{e}")


if __name__ == "__main__":
    main()
```

运行：

```bash
python todo.py add "买牛奶"    # 添加
python todo.py list            # 查看
python todo.py done 1          # 标记完成
python todo.py remove 1        # 删除
```

**用到**：`class`、`@装饰器`、生成器（`sum(... for ...)`）、`lambda`、`argparse`、`json` 文件读写、`pathlib`、异常处理、`if __name__`。
